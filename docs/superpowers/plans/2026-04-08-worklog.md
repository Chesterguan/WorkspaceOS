import uuid

from sqlalchemy.dialects.postgresql import ARRAY, JSONB, UUID
from sqlalchemy.orm import Mapped, mapped_column

from app.database import Base


class WorkLog(Base):
    __tablename__ = "work_logs"

    id: Mapped[uuid.UUID] = mapped_column(
        UUID(as_uuid=True), primary_key=True, server_default=func.gen_random_uuid(),

    title: Mapped[str] = mapped_column(String(500), nullable=False)

"""
Work Log service: data gathering, AI report generation, DOCX export.

Pulls activity data from the database for a given time period and set of
projects, generates a structured progress report via AI, and exports to
DOCX with tables and embedded charts.
"""
import base64
import io
import logging
import re
import uuid
from datetime import date, datetime
from typing import Any, Dict, List, Optional

from sqlalchemy import func, select, text
from sqlalchemy.ext.asyncio import AsyncSession

from app.models.project import Project
from app.services.ai_client import get_cloud_client

logger = logging.getLogger(__name__)

async def gather_period_data(
    project_ids: List[uuid.UUID],
    start_date: date,
    end_date: date,
    db: AsyncSession,
) -> Dict[str, Any]:
    """Query all activity data for the given projects and time period."""
    pid_list = [str(p) for p in project_ids]
    pid_sql = ",".join(f"'{p}'" for p in pid_list)

    # Project names
    proj_result = await db.execute(
        select(Project).where(Project.id.in_(project_ids))
    )
    projects = {str(p.id): p.name for p in proj_result.scalars().all()}

    # Commits by project
    commits_query = text(f"""
        SELECT project_id::text, COUNT(*) as cnt
        FROM github_commits
        WHERE project_id IN ({pid_sql})
          AND committed_at >= :start AND committed_at < :end
        GROUP BY project_id
    """)
    commits_result = await db.execute(commits_query, {"start": start_date, "end": end_date})
    commits_by_project = {str(r[0]): int(r[1]) for r in commits_result.all()}

    # Commits by week (for charts)
    weekly_query = text(f"""
        SELECT date_trunc('week', committed_at)::date as week, COUNT(*)
        FROM github_commits
        WHERE project_id IN ({pid_sql})
          AND committed_at >= :start AND committed_at < :end
        GROUP BY week ORDER BY week
    """)
    weekly_result = await db.execute(weekly_query, {"start": start_date, "end": end_date})
    weekly_commits = [{"week": str(r[0]), "count": int(r[1])} for r in weekly_result.all()]

    # Papers (blog_posts with paper tag)
    papers_query = text(f"""
        SELECT project_id::text, title, created_at::text
        FROM blog_posts
        WHERE project_id IN ({pid_sql})
          AND tags @> ARRAY['paper']::text[]
          AND created_at >= :start AND created_at < :end
        ORDER BY created_at
    """)
    papers_result = await db.execute(papers_query, {"start": start_date, "end": end_date})
    papers = [{"project_id": r[0], "title": r[1], "date": r[2]} for r in papers_result.all()]

    # Drafts by project and status
    drafts_query = text(f"""
        SELECT project_id::text, status, COUNT(*)
        FROM drafts
        WHERE project_id IN ({pid_sql})
          AND created_at >= :start AND created_at < :end
        GROUP BY project_id, status
    """)
    drafts_result = await db.execute(drafts_query, {"start": start_date, "end": end_date})
    drafts = [{"project_id": r[0], "status": r[1], "count": int(r[2])} for r in drafts_result.all()]

    # Sync runs
    syncs_query = text(f"""
        SELECT project_id::text, COUNT(*), COALESCE(SUM(commits_fetched), 0)
        FROM sync_runs
        WHERE project_id IN ({pid_sql})
          AND completed_at >= :start AND completed_at < :end
          AND status = 'completed'
        GROUP BY project_id
    """)
    syncs_result = await db.execute(syncs_query, {"start": start_date, "end": end_date})
    syncs = {r[0]: {"runs": int(r[1]), "commits_fetched": int(r[2])} for r in syncs_result.all()}

    total_commits = sum(commits_by_project.values())
    total_papers = len(papers)
    total_drafts = sum(d["count"] for d in drafts)

    return {
        "projects": projects,
        "commits_by_project": commits_by_project,
        "weekly_commits": weekly_commits,
        "papers": papers,
        "drafts": drafts,
        "syncs": syncs,
        "totals": {
            "commits": total_commits,
            "papers": total_papers,
            "drafts": total_drafts,
            "projects": len(project_ids),
        },
        "period": {
            "start": str(start_date),
            "end": str(end_date),
        },
    }

async def generate_report(
    period_type: str,
    period_data: Dict,
    goals: List[Dict],
    additional_instructions: Optional[str] = None,
) -> str:
    """Generate a progress report using cloud AI."""
    template = _REPORT_TEMPLATES.get(period_type, _REPORT_TEMPLATES["weekly"])

    # Format goals for the prompt
    goals_text = ""
    if goals:
        goals_lines = []
        for g in goals:
            status = "Done" if g.get("achieved") else "Pending"
            goals_lines.append(f"- {g['description']} [{status}] — {g.get('evidence', 'no evidence yet')}")
        goals_text = "Goals set for this period:\n" + "\n".join(goals_lines)

    # Format project data
    projects = period_data.get("projects", {})
    commits = period_data.get("commits_by_project", {})
    project_lines = []
    for pid, name in projects.items():
        c = commits.get(pid, 0)
        project_lines.append(f"- {name}: {c} commits")

    data_context = (
        f"Period: {period_data['period']['start']} to {period_data['period']['end']}\n"
        f"Projects:\n" + "\n".join(project_lines) + "\n\n"
        f"Totals: {period_data['totals']['commits']} commits, "
        f"{period_data['totals']['papers']} papers, "
        f"{period_data['totals']['drafts']} drafts\n\n"
    )

    if period_data.get("papers"):
        data_context += "Papers generated:\n"
        for p in period_data["papers"]:
            pname = projects.get(p["project_id"], "Unknown")
            data_context += f"- [{pname}] {p['title']}\n"

    if goals_text:
        data_context += f"\n{goals_text}\n"

    if additional_instructions:
        data_context += f"\nAdditional instructions: {additional_instructions}\n"

    ai = get_cloud_client()
    user_prompt = f"## Activity Data\n\n{data_context}\n\n---\nGenerate the report now."

    try:
        content = await ai.complete(system=template, user=user_prompt)
    except Exception:
        logger.exception("worklog: AI report generation failed")
        content = f"## Report Generation Failed\n\nCould not generate report. Please try again."

    return content

def _generate_chart_png(weekly_data: List[Dict]) -> Optional[bytes]:
    """Generate a bar chart of weekly commits as PNG bytes."""
    if not weekly_data:
        return None

    try:
        import matplotlib
        matplotlib.use("Agg")  # non-interactive backend
        import matplotlib.pyplot as plt

        weeks = [d["week"][-5:] for d in weekly_data]  # "MM-DD" format
        counts = [d["count"] for d in weekly_data]

        fig, ax = plt.subplots(figsize=(6, 2.5))
        ax.bar(weeks, counts, color="#10B981", width=0.6)
        ax.set_ylabel("Commits", fontsize=9)
        ax.set_title("Weekly Activity", fontsize=10, fontweight="bold")
        ax.tick_params(axis="both", labelsize=8)
        plt.tight_layout()

        buf = io.BytesIO()
        fig.savefig(buf, format="png", dpi=150)
        plt.close(fig)
        buf.seek(0)
        return buf.getvalue()
    except Exception:
        logger.exception("worklog: chart generation failed")
        return None


def export_to_docx(
    content: str,
    title: str,
    period_data: Optional[Dict] = None,
) -> bytes:
    """Convert a markdown report to a DOCX file with tables and charts."""
    from docx import Document
    from docx.shared import Inches, Pt
    from docx.enum.text import WD_ALIGN_PARAGRAPH

    doc = Document()

    # Title
    title_para = doc.add_heading(title, level=0)
    title_para.alignment = WD_ALIGN_PARAGRAPH.CENTER

    # Period subtitle
    if period_data and period_data.get("period"):
        sub = doc.add_paragraph(
            f"{period_data['period']['start']} to {period_data['period']['end']}"
        )
        sub.alignment = WD_ALIGN_PARAGRAPH.CENTER
        sub.runs[0].font.size = Pt(10)
        sub.runs[0].font.italic = True

    doc.add_paragraph("")  # spacer

    # Parse markdown and convert
    lines = content.split("\n")
    i = 0
    while i < len(lines):
        line = lines[i]

        # Headings
        if line.startswith("## "):
            doc.add_heading(line[3:].strip(), level=2)
        elif line.startswith("### "):
            doc.add_heading(line[4:].strip(), level=3)

        # Table detection (markdown pipe tables)
        elif "|" in line and i + 1 < len(lines) and "---" in lines[i + 1]:
            # Parse markdown table
            header_cells = [c.strip() for c in line.split("|") if c.strip()]
            i += 1  # skip separator line

            rows_data = []
            while i + 1 < len(lines) and "|" in lines[i + 1]:
                i += 1
                row_cells = [c.strip() for c in lines[i].split("|") if c.strip()]
                rows_data.append(row_cells)

            # Create Word table
            if header_cells:
                table = doc.add_table(rows=1 + len(rows_data), cols=len(header_cells))
                table.style = "Light Grid Accent 1"

                # Header row
                for j, cell_text in enumerate(header_cells):
                    table.rows[0].cells[j].text = cell_text

                # Data rows
                for row_idx, row_data in enumerate(rows_data):
                    for j, cell_text in enumerate(row_data):
                        if j < len(table.columns):
                            table.rows[row_idx + 1].cells[j].text = cell_text

                doc.add_paragraph("")  # spacer after table

        # Bullet list
        elif line.startswith("- "):
            text = line[2:].strip()
            # Bold text within **...**
            para = doc.add_paragraph(style="List Bullet")
            _add_formatted_text(para, text)

        # Regular text
        elif line.strip():
            para = doc.add_paragraph()
            _add_formatted_text(para, line.strip())

        i += 1

    # Embed chart for monthly/quarterly reports
    if period_data and period_data.get("weekly_commits"):
        chart_png = _generate_chart_png(period_data["weekly_commits"])
        if chart_png:
            doc.add_heading("Activity Chart", level=2)
            image_stream = io.BytesIO(chart_png)
            doc.add_picture(image_stream, width=Inches(5))

    # Save to bytes
    output = io.BytesIO()
    doc.save(output)
    output.seek(0)
    return output.getvalue()


def _add_formatted_text(para, text: str) -> None:
    """Add text with basic markdown formatting (**bold**, *italic*) to a paragraph."""
    # Split on **bold** markers
    parts = re.split(r"(\*\*.*?\*\*)", text)
    for part in parts:
        if part.startswith("**") and part.endswith("**"):
            run = para.add_run(part[2:-2])
            run.bold = True
        else:
            para.add_run(part)

"""Work Log router: generate, list, update, export progress reports."""
import base64
import uuid
from datetime import date, datetime

from fastapi import APIRouter, Depends, HTTPException, Query, status
from sqlalchemy import select, func
from sqlalchemy.ext.asyncio import AsyncSession

from app.dependencies import get_db, verify_api_key
from app.models.worklog import WorkLog
from app.schemas.worklog import (
    ExportDocxResponse,
    GenerateWorkLogRequest,
    UpdateWorkLogRequest,
    WorkLogListResponse,
    WorkLogResponse,
)
from app.services import worklog_service

router = APIRouter(prefix="/worklog", tags=["worklog"])


def _to_response(wl: WorkLog) -> WorkLogResponse:
    return WorkLogResponse(
        id=wl.id,
        title=wl.title,
        period_type=wl.period_type,
        period_start=wl.period_start,
        period_end=wl.period_end,
        project_ids=wl.project_ids,
        content=wl.content,
        goals=wl.goals,
        created_at=wl.created_at,
        updated_at=wl.updated_at,
    )


@router.post(
    "/generate",
    response_model=WorkLogResponse,
    status_code=status.HTTP_201_CREATED,
    summary="Generate a new work log report",
)
async def generate_worklog(
    body: GenerateWorkLogRequest,
    db: AsyncSession = Depends(get_db),
    _key: str = Depends(verify_api_key),
) -> WorkLogResponse:
    """Generate a progress report from project data for the given period."""
    valid_types = ("weekly", "monthly", "quarterly")
    if body.period_type not in valid_types:
        raise HTTPException(status_code=422, detail=f"period_type must be one of: {valid_types}")

    start = date.fromisoformat(body.period_start)
    end = date.fromisoformat(body.period_end)

    # Gather data
    period_data = await worklog_service.gather_period_data(
        body.project_ids, start, end, db,
    )

    # Generate report
    goals_dicts = [g.model_dump() for g in body.goals]
    content = await worklog_service.generate_report(
        body.period_type, period_data, goals_dicts, body.additional_instructions,
    )

    # Build title
    type_label = body.period_type.capitalize()
    title = f"{type_label} Report: {body.period_start} to {body.period_end}"

    # Save
    wl = WorkLog(
        title=title,
        period_type=body.period_type,
        period_start=start,
        period_end=end,
        project_ids=body.project_ids,
        content=content,
        goals=goals_dicts if goals_dicts else None,
    )
    db.add(wl)
    await db.flush()
    await db.refresh(wl)

    return _to_response(wl)


@router.get("", response_model=WorkLogListResponse, summary="List saved reports")
async def list_worklogs(
    limit: int = Query(default=20, ge=1, le=100),
    offset: int = Query(default=0, ge=0),
    db: AsyncSession = Depends(get_db),
    _key: str = Depends(verify_api_key),
) -> WorkLogListResponse:
    """List all saved work log reports, newest first."""
    total_result = await db.execute(select(func.count(WorkLog.id)))
    total = total_result.scalar_one()

    result = await db.execute(
        select(WorkLog).order_by(WorkLog.created_at.desc()).limit(limit).offset(offset)
    )
    reports = [_to_response(wl) for wl in result.scalars().all()]
    return WorkLogListResponse(reports=reports, total=total)


@router.get("/{worklog_id}", response_model=WorkLogResponse, summary="Get a report")
async def get_worklog(
    worklog_id: uuid.UUID,
    db: AsyncSession = Depends(get_db),
    _key: str = Depends(verify_api_key),
) -> WorkLogResponse:
    result = await db.execute(select(WorkLog).where(WorkLog.id == worklog_id))
    wl = result.scalar_one_or_none()
    if wl is None:
        raise HTTPException(status_code=404, detail="Report not found")
    return _to_response(wl)


@router.put("/{worklog_id}", response_model=WorkLogResponse, summary="Update a report")
async def update_worklog(
    worklog_id: uuid.UUID,
    body: UpdateWorkLogRequest,
    db: AsyncSession = Depends(get_db),
    _key: str = Depends(verify_api_key),
) -> WorkLogResponse:
    result = await db.execute(select(WorkLog).where(WorkLog.id == worklog_id))
    wl = result.scalar_one_or_none()
    if wl is None:
        raise HTTPException(status_code=404, detail="Report not found")

    if body.content is not None:
        wl.content = body.content
    if body.goals is not None:
        wl.goals = [g.model_dump() for g in body.goals]
    if body.title is not None:
        wl.title = body.title

    await db.flush()
    await db.refresh(wl)
    return _to_response(wl)


@router.delete("/{worklog_id}", status_code=status.HTTP_204_NO_CONTENT, summary="Delete a report")
async def delete_worklog(
    worklog_id: uuid.UUID,
    db: AsyncSession = Depends(get_db),
    _key: str = Depends(verify_api_key),
) -> None:
    result = await db.execute(select(WorkLog).where(WorkLog.id == worklog_id))
    wl = result.scalar_one_or_none()
    if wl is None:
        raise HTTPException(status_code=404, detail="Report not found")
    await db.delete(wl)


@router.post("/{worklog_id}/export-docx", response_model=ExportDocxResponse, summary="Export to DOCX")
async def export_docx(
    worklog_id: uuid.UUID,
    db: AsyncSession = Depends(get_db),
    _key: str = Depends(verify_api_key),
) -> ExportDocxResponse:
    """Export a work log report to a Word document with tables and charts."""
    result = await db.execute(select(WorkLog).where(WorkLog.id == worklog_id))
    wl = result.scalar_one_or_none()
    if wl is None:
        raise HTTPException(status_code=404, detail="Report not found")

    # Re-gather data for charts
    period_data = await worklog_service.gather_period_data(
        wl.project_ids, wl.period_start, wl.period_end, db,
    )

    docx_bytes = worklog_service.export_to_docx(wl.content, wl.title, period_data)
    docx_b64 = base64.b64encode(docx_bytes).decode("utf-8")
    filename = f"{wl.title.replace(' ', '_').replace(':', '')}.docx"

    return ExportDocxResponse(docx_base64=docx_b64, filename=filename)
